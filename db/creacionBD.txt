create database robot_defender;
use robot_defender;

create table Player
(
    id int AUTO_INCREMENT,
    oleadaMax int null,
    tiempoMaximo float null,
    primary key(id)
);

create table Sessions
(
    id int AUTO_INCREMENT,
    fechaFin date null,
    oleada int not null,
    tiempoDefendido float not null,
    primary key(id)
);

create table TablaClasificacion
(
    id int AUTO_INCREMENT,
    PlayerFK int not null,
    SessionsFK int not null,
    
    primary key(id),
    foreign key(PlayerFK) references Player(id),
    foreign key(SessionsFK) references Sessions(id)
);

create table catalogoEnemigos
(
    id int AUTO_INCREMENT,
    SessionsFK int not null,
    primary key(id),
    foreign key(SessionsFK) references Sessions(id) 
);

create table EnemyType
(
    id int AUTO_INCREMENT,
    velocidad float not null,
    nombre varchar(20) not null,
    catalogoEnemigosFK int not null,
    primary key(id),
    foreign key(catalogoEnemigosFK) references catalogoEnemigos(id) 
);

create table EnemyInstance
(
    id int AUTO_INCREMENT,
    velocidadDisparo float not null,
    x float not null,
    y float not null,
    EnemyTypeFK int not null,
    primary key(id),
    foreign key(EnemyTypeFK) references EnemyType(id) 
);

create table CombatLogs
(
    id int AUTO_INCREMENT,
    tipoAccion varchar(50) not null,
    origen varchar(50) not null,
    destino varchar(50) not null,
    stamptime float not null,
    SessionsFK int not null,
    primary key(id),
    foreign key(SessionsFK) references Sessions(id) 
);

create table SessionResults
(
    id int AUTO_INCREMENT,
    tiempoMuerte float not null,
    motivoFin varchar(50) not null,
    enemigosDestruidos int not null,
	SessionsFK int not null,
    primary key(id),
    foreign key(SessionsFK) references Sessions(id) 
);



USE robot_defender;

DELIMITER $$

DROP PROCEDURE IF EXISTS CrearPlayer$$
CREATE PROCEDURE CrearPlayer(
	IN p_id INT,
	IN p_oleadaMax INT,
	IN p_tiempoDefendido FLOAT
)
BEGIN
	INSERT INTO Player (id, oleadaMax, tiempoMaximo)
	VALUES (p_id, p_oleadaMax, p_tiempoDefendido)
	ON DUPLICATE KEY UPDATE
		oleadaMax = COALESCE(p_oleadaMax, oleadaMax),
		tiempoMaximo = COALESCE(p_tiempoDefendido, tiempoMaximo);
END$$

DROP PROCEDURE IF EXISTS CrearSessions$$
CREATE PROCEDURE CrearSessions(
	IN p_id INT,
	IN p_fechaFin DATE,
	IN p_oleada INT,
	IN p_tiempoDefendido FLOAT
)
BEGIN
	INSERT INTO Sessions (id, fechaFin, oleada, tiempoDefendido)
	VALUES (p_id, p_fechaFin, p_oleada, p_tiempoDefendido)
	ON DUPLICATE KEY UPDATE
		fechaFin = p_fechaFin,
		oleada = p_oleada,
		tiempoDefendido = p_tiempoDefendido;
END$$

DROP PROCEDURE IF EXISTS ActualizarSessions$$
CREATE PROCEDURE ActualizarSessions(
	IN p_id INT,
	IN p_fechaFin DATE,
	IN p_oleada INT,
	IN p_tiempoDefendido FLOAT,
	IN p_catalogoEnemigosFK INT
)
BEGIN
	-- Se conserva p_catalogoEnemigosFK por compatibilidad con ConexionBD.
	-- La tabla Sessions del esquema no contiene esa columna.
	UPDATE Sessions
	SET fechaFin = p_fechaFin,
		oleada = p_oleada,
		tiempoDefendido = p_tiempoDefendido
	WHERE id = p_id;
END$$

DROP PROCEDURE IF EXISTS CrearTablaClasificacion$$
CREATE PROCEDURE CrearTablaClasificacion(
	IN p_id INT,
	IN p_playerFK INT,
	IN p_sessionsFK INT
)
BEGIN

	VALUES (p_id, p_playerFK, p_sessionsFK)
	ON DUPLICATE KEY UPDATE
		PlayerFK = p_playerFK,
		SessionsFK = p_sessionsFK;
END$$

DROP PROCEDURE IF EXISTS CrearCatalogoEnemigos$$
CREATE PROCEDURE CrearCatalogoEnemigos(
	IN p_id INT,
	IN p_sessionsFK INT
)
BEGIN
	INSERT INTO catalogoEnemigos (id, SessionsFK)
	VALUES (p_id, p_sessionsFK)
	ON DUPLICATE KEY UPDATE SessionsFK = p_sessionsFK;
END$$

DROP PROCEDURE IF EXISTS CrearEnemyType$$

	IN p_origen VARCHAR(50),
	IN p_destino VARCHAR(50),
	IN p_stamptime FLOAT,
	IN p_sessionsFK INT
)
BEGIN
	INSERT INTO CombatLogs (id, tipoAccion, origen, destino, stamptime, SessionsFK)
	VALUES (p_id, p_tipoAccion, p_origen, p_destino, p_stamptime, p_sessionsFK)
	ON DUPLICATE KEY UPDATE
		tipoAccion = p_tipoAccion,
		origen = p_origen,
		destino = p_destino,
		stamptime = p_stamptime,

DROP PROCEDURE IF EXISTS CrearSessionResults$$
CREATE PROCEDURE CrearSessionResults(
	IN p_id INT,

	IN p_sessionsFK INT

BEGIN
	INSERT INTO SessionResults (id, tiempoMuerte, motivoFin, enemigosDestruidos, SessionsFK)
	VALUES (p_id, p_tiempoMuerte, p_motivoFin, p_enemigosDestruidos, p_sessionsFK)
	ON DUPLICATE KEY UPDATE
		tiempoMuerte = p_tiempoMuerte,
		motivoFin = p_motivoFin,
		enemigosDestruidos = p_enemigosDestruidos,
		SessionsFK = p_sessionsFK;
END$$

DROP PROCEDURE IF EXISTS EliminarTablaClasificacion$$
CREATE PROCEDURE EliminarTablaClasificacion(IN p_id INT)
BEGIN
	DELETE FROM TablaClasificacion WHERE id = p_id;
END$$

DROP PROCEDURE IF EXISTS EliminarEnemyType$$
CREATE PROCEDURE EliminarEnemyType(IN p_id INT)
BEGIN
	DELETE FROM EnemyInstance WHERE EnemyTypeFK = p_id;
	DELETE FROM EnemyType WHERE id = p_id;
END$$

DROP PROCEDURE IF EXISTS EliminarCatalogoEnemigos$$
CREATE PROCEDURE EliminarCatalogoEnemigos(IN p_id INT)
BEGIN
	DELETE FROM EnemyInstance
	WHERE EnemyTypeFK IN (
		SELECT id FROM EnemyType WHERE catalogoEnemigosFK = p_id
	);
	DELETE FROM EnemyType WHERE catalogoEnemigosFK = p_id;
	DELETE FROM catalogoEnemigos WHERE id = p_id;
END$$

DROP PROCEDURE IF EXISTS EliminarSessions$$
CREATE PROCEDURE EliminarSessions(IN p_id INT)
BEGIN
	DELETE FROM CombatLogs WHERE SessionsFK = p_id;
	DELETE FROM SessionResults WHERE SessionsFK = p_id;
	DELETE FROM TablaClasificacion WHERE SessionsFK = p_id;

	DELETE FROM EnemyInstance
	WHERE EnemyTypeFK IN (
		SELECT et.id
		FROM EnemyType AS et
		INNER JOIN catalogoEnemigos AS ce
			ON ce.id = et.catalogoEnemigosFK
		WHERE ce.SessionsFK = p_id
	);

	DELETE et
	FROM EnemyType AS et
	INNER JOIN catalogoEnemigos AS ce
		ON ce.id = et.catalogoEnemigosFK
	WHERE ce.SessionsFK = p_id;

	DELETE FROM catalogoEnemigos WHERE SessionsFK = p_id;
	DELETE FROM Sessions WHERE id = p_id;
END$$

DROP PROCEDURE IF EXISTS EliminarPlayer$$
CREATE PROCEDURE EliminarPlayer(IN p_id INT)
BEGIN
	DELETE FROM TablaClasificacion WHERE PlayerFK = p_id;
	DELETE FROM Player WHERE id = p_id;
END$$
